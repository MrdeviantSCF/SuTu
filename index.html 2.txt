<script>
/* =========================================================
   INTERACTION ENGINE
   ========================================================= */
(() => {
  "use strict";

  const reduced = window.matchMedia("(prefers-reduced-motion: reduce)").matches;

  /* ---------- AUDIO & GATE ---------- */
  const gate = document.getElementById("gate");
  const openBtn = document.getElementById("openBtn");
  const music = document.getElementById("bgMusic");
  const musicBtn = document.getElementById("musicBtn");
  const musicIcon = document.getElementById("musicIcon");
  const musicLabel = document.getElementById("musicLabel");

  music.volume = 0.16;
  let fadeTimer = null;

  function fadeMusicTo(target, duration = 2200){
    clearInterval(fadeTimer);
    const start = music.volume;
    const steps = 44;
    const delta = (target - start) / steps;
    let step = 0;

    fadeTimer = setInterval(() => {
      step++;
      music.volume = Math.max(0, Math.min(0.20, start + delta * step));
      if(step >= steps){
        clearInterval(fadeTimer);
        music.volume = target;
      }
    }, duration / steps);
  }

  function setMusicUI(on){
    musicBtn.classList.toggle("playing", on);
    musicIcon.textContent = on ? "♫" : "♪";
    musicLabel.textContent = on ? "MUSIC ON" : "MUSIC OFF";
  }

  async function startMusic(){
    try{
      music.volume = 0.01;
      await music.play();
      setMusicUI(true);
      fadeMusicTo(0.16, 2800);
    }catch(err){
      setMusicUI(false);
    }
  }

  function stopMusic(){
    fadeMusicTo(0, 700);
    setTimeout(() => {
      music.pause();
      music.currentTime = 0;
    }, 760);
    setMusicUI(false);
  }

  openBtn.addEventListener("click", () => {
    gate.classList.add("hide");
    document.body.classList.remove("lock");
    startMusic();

    setTimeout(() => {
      document.getElementById("hero").scrollIntoView({
        behavior: reduced ? "auto" : "smooth"
      });
    }, 180);
  });

  musicBtn.addEventListener("click", async () => {
    if(music.paused){
      startMusic();
    }else{
      stopMusic();
    }
  });

  window.addEventListener("keydown", e => {
    if ((e.key === "Enter" || e.key === " ") && !gate.classList.contains("hide")) {
      e.preventDefault();
      openBtn.click();
    }
  });

  /* ---------- PHOTO CAROUSEL ---------- */
  const slides = document.querySelectorAll(".slideImg");
  const dots = document.querySelectorAll(".dot");
  const prevBtn = document.getElementById("prevSlide");
  const nextBtn = document.getElementById("nextSlide");
  let currentSlide = 0;
  let autoSlideTimer = null;

  function showSlide(index) {
    slides[currentSlide].classList.remove("active");
    dots[currentSlide].classList.remove("active");
    currentSlide = (index + slides.length) % slides.length;
    slides[currentSlide].classList.add("active");
    dots[currentSlide].classList.add("active");
  }

  function resetAutoSlide() {
    clearInterval(autoSlideTimer);
    autoSlideTimer = setInterval(() => {
      showSlide(currentSlide + 1);
    }, 3500);
  }

  if(prevBtn && nextBtn){
    prevBtn.addEventListener("click", () => {
      showSlide(currentSlide - 1);
      resetAutoSlide();
    });
    nextBtn.addEventListener("click", () => {
      showSlide(currentSlide + 1);
      resetAutoSlide();
    });
    dots.forEach((dot, i) => {
      dot.addEventListener("click", () => {
        showSlide(i);
        resetAutoSlide();
      });
    });
    resetAutoSlide();
  }

  /* ---------- REVEAL ON SCROLL ---------- */
  const revealObserver = new IntersectionObserver(
    entries => {
      entries.forEach(entry => {
        if (entry.isIntersecting) {
          entry.target.classList.add("show");
        }
      });
    },
    { threshold: 0.12, rootMargin: "0px 0px -7% 0px" }
  );

  document.querySelectorAll(".reveal").forEach(el => revealObserver.observe(el));

  /* ---------- CURSOR GLOW ---------- */
  const glow = document.getElementById("cursorGlow");
  if (!reduced && window.matchMedia("(hover:hover)").matches) {
    window.addEventListener("pointermove", e => {
      glow.style.left = e.clientX + "px";
      glow.style.top = e.clientY + "px";
    }, { passive: true });
  }

  /* ---------- PROGRESS ---------- */
  const sections = [...document.querySelectorAll("main > section")];
  const progressNo = document.getElementById("progressNo");
  const progressFill = document.getElementById("progressFill");

  const sectionObserver = new IntersectionObserver(
    entries => {
      entries.forEach(entry => {
        if (entry.isIntersecting) {
          const index = sections.indexOf(entry.target) + 1;
          progressNo.textContent = String(index).padStart(2, "0");
        }
      });
    },
    { threshold: 0.52 }
  );

  sections.forEach(section => sectionObserver.observe(section));

  function updateProgress() {
    const max = document.documentElement.scrollHeight - window.innerHeight;
    const percent = max > 0 ? Math.min(100, Math.max(0, window.scrollY / max * 100)) : 0;
    progressFill.style.height = percent + "%";
  }

  window.addEventListener("scroll", updateProgress, { passive: true });
  updateProgress();

  /* ---------- FINAL THOUGHT ---------- */
  const thoughtBtn = document.getElementById("thoughtBtn");
  const hiddenThought = document.getElementById("hiddenThought");

  thoughtBtn.addEventListener("click", () => {
    hiddenThought.classList.toggle("open");
    thoughtBtn.textContent = hiddenThought.classList.contains("open")
      ? "Keep this thought ↗"
      : "One last thought →";
  });

  /* ---------- STAR FIELD ---------- */
  const canvas = document.getElementById("stars");
  const ctx = canvas.getContext("2d");
  let stars = [];

  function resizeStars() {
    const dpr = Math.min(window.devicePixelRatio || 1, 2);
    canvas.width = window.innerWidth * dpr;
    canvas.height = window.innerHeight * dpr;
    canvas.style.width = window.innerWidth + "px";
    canvas.style.height = window.innerHeight + "px";
    ctx.setTransform(dpr, 0, 0, dpr, 0, 0);

    const count = window.innerWidth < 700 ? 45 : 90;
    stars = Array.from({ length: count }, () => ({
      x: Math.random() * window.innerWidth,
      y: Math.random() * window.innerHeight,
      r: Math.random() * 1.15 + .15,
      a: Math.random() * .42 + .08,
      speed: Math.random() * .16 + .02,
      phase: Math.random() * Math.PI * 2
    }));
  }

  function drawStars(time = 0) {
    ctx.clearRect(0, 0, window.innerWidth, window.innerHeight);

    stars.forEach(star => {
      const pulse = star.a + Math.sin(time * .001 + star.phase) * .06;
      ctx.beginPath();
      ctx.arc(star.x, star.y, star.r, 0, Math.PI * 2);
      ctx.fillStyle = `rgba(255,235,245,${Math.max(.03, pulse)})`;
      ctx.fill();

      star.y -= star.speed;
      if (star.y < -5) star.y = window.innerHeight + 5;
    });

    if (!reduced) requestAnimationFrame(drawStars);
  }

  resizeStars();
